class m_agent extends uvm_agent;
  `uvm_component_utils(m_agent)
  apb_drv drv;
  apb_m_mon mon;
  apb_seqr seqr;
  apb_config cfg;
  
  function new(string name="m_agent",uvm_component parent);
    super.new(name,parent);
  endfunction
  
  function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    if(!uvm_config_db#(apb_config)::get(this,"","apb_config",cfg))
      `uvm_fatal(get_type_name(),"cannot get cfg")
    mon=apb_m_mon::type_id::create("mon",this);
    if(cfg.m_is_active == UVM_ACTIVE)
      begin
        seqr=apb_seqr::type_id::create("seqr",this);
        drv=apb_drv::type_id::create("drv",this);
      end
  endfunction
  
  function void connect_phase(uvm_phase phase);
    if(cfg.m_is_active == UVM_ACTIVE)
      begin
        drv.seq_item_port.connect(seqr.seq_item_export);
      end
  endfunction
  
endclass
