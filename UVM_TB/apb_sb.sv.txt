  uvm_tlm_analysis_fifo#(apb_trans) fifo_rdh;
  
  apb_trans xtn1,xtn2;
  function new(string name,uvm_component parent);
    super.new(name,parent);
    fifo_wrh=new("fifo_wrh",this);
    fifo_rdh=new("fifo_rdh",this);
  endfunction
  
  function void build_phase(uvm_phase phase);

    xtn1=apb_trans::type_id::create("xtn1");
    xtn2=apb_trans::type_id::create("xtn2");
  endfunction
  
  task run_phase(uvm_phase phase);
    fork
    	forever
          		begin

                  `uvm_info(get_type_name(),"data received from slave_monitor",UVM_MEDIUM)
