  class apb_drv extends uvm_driver#(apb_trans);
  `uvm_component_utils(apb_drv)
  
  virtual apb_if.DR_MP vif;
  
  apb_config cfg;
  
  function new(string name="apb_drv",uvm_component parent);
    super.new(name,parent);
  endfunction
  
  function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    
    if(!uvm_config_db#(apb_config)::get(this,"","apb_config",cfg))
      `uvm_fatal(get_type_name(),"cannot get config")
  endfunction
  
  function void connect_phase(uvm_phase phase);
    vif = cfg.vif;
  endfunction
  
  task run_phase(uvm_phase phase);
    vif.drv_cb.Psel     <= 0;
    vif.drv_cb.Penable  <= 0;
    vif.drv_cb.Pwrite   <= 0;
    vif.drv_cb.Paddr    <= 0;
    vif.drv_cb.Pwdata   <= 0;
    
    @(vif.drv_cb)
    vif.drv_cb.Presetn<=0;
    @(vif.drv_cb)
    vif.drv_cb.Presetn<=1;
    
    forever begin
      seq_item_port.get_next_item(req);
      send_to_dut(req);
      seq_item_port.item_done();
    end
  endtask
  
  task send_to_dut(apb_trans xtn);
   if(xtn.Pwrite==1)
    begin
      @(vif.drv_cb);
    vif.drv_cb.Paddr<=xtn.Paddr;
    vif.drv_cb.Pwrite<=xtn.Pwrite;
    vif.drv_cb.Psel<=xtn.Psel;
    vif.drv_cb.Pwdata<=xtn.Pwdata;
    vif.drv_cb.Penable<=0;
//      repeat(2)
    @(vif.drv_cb);
    vif.drv_cb.Penable<=1;
      
@(vif.drv_cb);
      wait(vif.drv_cb.Pready)
 //       @(vif.drv_cb);
      
 //   @(vif.drv_cb);
      vif.drv_cb.Penable<=0;
	  vif.drv_cb.Psel<=0;
//    @(vif.drv_cb);
    end
    
    else begin
      @(vif.drv_cb);
      vif.drv_cb.Penable<=0;
      vif.drv_cb.Paddr<=xtn.Paddr;
      
  	  vif.drv_cb.Pwrite<=xtn.Pwrite;
  	  vif.drv_cb.Psel<=xtn.Psel;
    @(vif.drv_cb);
    vif.drv_cb.Penable<=1;
      
     @(vif.drv_cb);
      wait(vif.drv_cb.Pready)
 //       @(vif.drv_cb);
      
 //   @(vif.drv_cb);
      vif.drv_cb.Penable<=0;
	  vif.drv_cb.Psel<=0;
//    @(vif.drv_cb);
    end
    

    
    `uvm_info("MASTER DRIVER",xtn.sprint,UVM_MEDIUM)
    
//    xtn.print();
  endtask
  
endclass
