	class apb_m_mon extends uvm_monitor;
  `uvm_component_utils(apb_m_mon)
  
  virtual apb_if.IMON_MP vif;
  
  apb_trans xtn;
  apb_config cfg;
  uvm_analysis_port#(apb_trans) monitor_port;
  
  function new(string name="apb_m_mon",uvm_component parent);
    super.new(name,parent);
    monitor_port=new("monitor_port",this);
  endfunction
  
  function void build_phase(uvm_phase phase);
    super.build_phase (phase);
    
    if(!uvm_config_db#(apb_config)::get(this,"","apb_config",cfg))
      `uvm_fatal(get_type_name(),"cannot get cfg")
  endfunction
  
  function void connect_phase (uvm_phase phase);
    vif=cfg.vif;
  endfunction
  
  task run_phase(uvm_phase phase);
 
    forever
          
      collect_data();
  endtask
  
  task collect_data();
      xtn=apb_trans::type_id::create("xtn");
 //   repeat(2) 
    @(vif.imon_cb)

    if(vif.imon_cb.Penable && vif.imon_cb.Psel && vif.imon_cb.Pwrite) 
      wait(vif.imon_cb.Pready)
    begin
      xtn.Paddr  = vif.imon_cb.Paddr;
        xtn.Pwdata = vif.imon_cb.Pwdata;
        xtn.Pwrite = vif.imon_cb.Pwrite;
        xtn.Psel   = vif.imon_cb.Psel;
        xtn.Penable= vif.imon_cb.Penable;
    
    `uvm_info("MASTER MONITOR",xtn.sprint,UVM_MEDIUM)
    
 //   xtn.print();
    monitor_port.write(xtn);
      end
  endtask
  
endclass
