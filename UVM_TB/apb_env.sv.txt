class apb_env extends uvm_env;
  `uvm_component_utils(apb_env)
  
  m_agent m_agt;
  s_agent s_agt;
  apb_sb sb;
  apb_config cfg;
  
  function new(string name="apb_env",uvm_component parent);
    super.new(name,parent);
  endfunction
  
  function void build_phase(uvm_phase phase);
    super.build_phase(phase);
    
    if(!uvm_config_db#(apb_config)::get(this,"","apb_config",cfg))
      `uvm_fatal(get_type_name(),"cannot get cfg")
    
    if(cfg.has_magent)begin
      m_agt=m_agent::type_id::create("m_agt",this);
    end
    
    if(cfg.has_sagent)begin
      s_agt=s_agent::type_id::create("s_agt",this);
    end
    
    if(cfg.has_scoreboard)begin
      sb=apb_sb::type_id::create("apb_sb",this);
    end
  endfunction
  
  function void connect_phase(uvm_phase phase);
    
    if(cfg.has_magent && cfg.has_scoreboard)begin
      
      m_agt.mon.monitor_port.connect(sb.fifo_wrh.analysis_export);
    end
    
  if(cfg.has_sagent && cfg.has_scoreboard)begin
      
    s_agt.mon.monitor_port.connect(sb.fifo_rdh.analysis_export);
    end
  endfunction
endclass
